# Runtime dependencies. Until now these existed only inside .venv, which is a darwin build
# and gitignored, so the dependency set was discoverable nowhere and could not be rebuilt.
#
# Versions are pinned rather than floating. The NAS resolves these fresh at image build,
# months after they were last exercised here, and an unpinned solver or HTTP client picking
# up a new major on a Tuesday is exactly the kind of failure a scheduled job reports to
# nobody. Bump deliberately, re-run the backtest, commit the new pin.

requests==2.34.2

# The LP solver. Ships its own CBC binary; on x86_64 (DS220+ is a Celeron J4025) that works
# as-is, with no coinor-cbc package and no solver path override.
pulp==3.3.2

# Imported unconditionally at the top of Marstek-planning.py even though the MQTT path is
# unused here, so the import fails without it and nothing runs.
paho-mqtt==2.1.0

# Not imported by name anywhere - it is the data behind zoneinfo. python:3.12-slim has no
# system tzdata, so ZoneInfo("Europe/Amsterdam") raises ZoneInfoNotFoundError without it.
# Every date and hour in this program is Amsterdam wall-clock; see the "Wall clock" block in
# Marstek-planning.py for why UTC would silently shorten the 14:05 run's horizon.
tzdata==2026.3
